procedure Tfrme01.Button1Click(Sender: TObject);
var INT: Integer;
begin
     if TryStrToInt(EditName.Text, INT) then
     with SQLQuery1 do
  begin
       if not Locate('商品名稱',EditName.Text,[loPartialkey]) then

        begin
        Append;
        fieldbyName('採購編號').asstring:='SRT9980';
        fieldbyName('商品名稱').asstring:=EditName.text;
        fieldbyName('採購日期').asstring:=EditDate.text;
        fieldbyName('出版商').asstring:=EditRelease.text;
        fieldbyName('數量').asinteger:=strToint(EditNumber.text);
        fieldbyName('單價').asinteger:=strToint(EditPrice.text);
        post;
        applyupdates;


        SQLTransaction1.CommitRetaining;
        refresh;
        showmessage('新商品新增成功');
        end
       else
        fieldbyName('數量').asstring:=EditNumber.text;
        showmessage('庫存數量更新');
  end
     else
      showmessage('項目不得為空白');
      SQLQuery1.Locate('商品名稱',EditName.Text,[loPartialkey]);
end;
